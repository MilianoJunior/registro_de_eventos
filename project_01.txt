    @desempenho
    def salvar_configuracao(self):
        """Salva a configuração de usinas e dispositivos"""
        try:
            config_data = request.get_json()

            if not config_data:
                return jsonify({"error": "Nenhum dado foi enviado"}), 400

            # Validação básica
            if not isinstance(config_data, dict):
                return jsonify({"error": "Formato de dados inválido"}), 400

            # Validar estrutura de cada usina
            for usina_nome, usina_data in config_data.items():
                if not isinstance(usina_data, dict):
                    return jsonify({"error": f"Dados inválidos para a usina {usina_nome}"}), 400

                # Verificar campos obrigatórios da usina
                if 'ip' not in usina_data or 'port' not in usina_data:
                    return jsonify({"error": f"IP e porta são obrigatórios para a usina {usina_nome}"}), 400

                # Verificar se tem pelo menos um dispositivo
                if 'dispositivos' not in usina_data or not usina_data['dispositivos']:
                    return jsonify({"error": f"A usina {usina_nome} precisa ter pelo menos um dispositivo"}), 400

                # Validar cada dispositivo
                for disp_nome, disp_data in usina_data['dispositivos'].items():
                    if not isinstance(disp_data, dict):
                        return jsonify({"error": f"Dados inválidos para o dispositivo {disp_nome}"}), 400

                    # Verificar campos obrigatórios do dispositivo
                    if 'conexao' not in disp_data:
                        return jsonify({"error": f"Conexão é obrigatória para o dispositivo {disp_nome}"}), 400

                    if 'ip' not in disp_data['conexao'] or 'port' not in disp_data['conexao']:
                        return jsonify({"error": f"IP e porta são obrigatórios para o dispositivo {disp_nome}"}), 400

            # Criar diretório se não existir
            os.makedirs(os.path.dirname(self.config_file_path), exist_ok=True)

            # Salvar configuração em arquivo JSON
            with open(self.config_file_path, 'w', encoding='utf-8') as f:
                json.dump(config_data, f, indent=2, ensure_ascii=False)

            return jsonify({
                "success": True,
                "message": "Configuração salva com sucesso",
                "usinas_count": len(config_data)
            }), 200

        except Exception as e:
            return jsonify({"error": str(e)}), 500

    @desempenho
    def carregar_configuracao(self):
        """Carrega a configuração salva"""
        try:
            if not os.path.exists(self.config_file_path):
                return jsonify({"error": "Nenhuma configuração encontrada"}), 404

            with open(self.config_file_path, 'r', encoding='utf-8') as f:
                config_data = json.load(f)

            return jsonify(config_data), 200

        except Exception as e:
            return jsonify({"error": str(e)}), 500

class Create:
    """
    Repositório de criação/inserção simples (parametrizado, reutilizável).
    """
    def __init__(
        self,
        tabela: str,
        db_instance: Optional[Database] = None,
    ):
        self.tabela = tabela
        self.db = db_instance or Database()

    def insert(self, data: Dict[str, Any]) -> Optional[int]:
        """
        Insere um novo registro na tabela.

        Args:
            data: Dicionário com os dados a serem inseridos (coluna: valor)

        Returns:
            ID do registro inserido ou None em caso de erro
        """
        try:
            self.db.connect()

            # Prepara as colunas e valores
            columns = list(data.keys())
            values = list(data.values())

            # Monta a query SQL
            placeholders = ", ".join(["%s"] * len(columns))
            columns_str = ", ".join([f"`{col}`" for col in columns])

            sql = f"INSERT INTO `{self.tabela}` ({columns_str}) VALUES ({placeholders})"

            # Executa a query
            cursor = self.db.execute_query(sql, tuple(values))

            if cursor:
                # Retorna o ID do último registro inserido
                return cursor.lastrowid

            return None

        except Exception as e:
            self._error('Create', 'insert', e)
            return None

    def insert_many(self, data_list: list[Dict[str, Any]]) -> Optional[int]:
        """
        Insere múltiplos registros de uma vez.

        Args:
            data_list: Lista de dicionários com os dados a serem inseridos

        Returns:
            Número de registros inseridos ou None em caso de erro
        """
        try:
            if not data_list:
                return 0

            self.db.connect()

            # Pega as colunas do primeiro registro
            columns = list(data_list[0].keys())
            columns_str = ", ".join([f"`{col}`" for col in columns])
            placeholders = ", ".join(["%s"] * len(columns))

            sql = f"INSERT INTO `{self.tabela}` ({columns_str}) VALUES ({placeholders})"

            # Prepara os valores de todos os registros
            values_list = [tuple(item[col] for col in columns) for item in data_list]

            # Executa a query para cada registro
            count = 0
            for values in values_list:
                cursor = self.db.execute_query(sql, values)
                if cursor:
                    count += 1

            return count

        except Exception as e:
            self._error('Create', 'insert_many', e)
            return None

    def _error(self, name, function, e):
        raise Exception(f"Erro: class: {name}, function: {function}: msg: {e}")

def get_connection_pool():

                        'connect_timeout': int(os.getenv('MYSQLCONNECTIONTIMEOUT', 10)),
                        'autocommit': False,

                    }

                    _connection_pool = pooling.MySQLConnectionPool(**pool_config)

                    if DB_DEBUG:
                        print(f"[ConnectionPool] ✅ Pool criado com {pool_config['pool_size']} conexões")

                except Error as e:

class _Singleton(type):
    _instance = None
    _lock = Lock()
    _created_at = None
    _uuid = None
    _create_count = 0  # quantas vezes tentou criar

    def __call__(cls, *args, **kwargs):
        cls._create_count += 1
        if cls._instance is None:
            with cls._lock:
                if cls._instance is None:
                    inst = super().__call__(*args, **kwargs)
                    cls._instance = inst
                    cls._created_at = time.strftime("%Y-%m-%d %H:%M:%S")
                    cls._uuid = str(uuid.uuid4())[:8]
                    if DB_DEBUG:
                        print(f"[Database] ✅ criado (uuid={cls._uuid}, pid={os.getpid()}, created_at={cls._created_at})")
                elif DB_DEBUG:
                    print(f"[Database] ♻️ reuso após corrida (uuid={cls._uuid})")
        else:
            if DB_DEBUG:
                print(f"[Database] ♻️ reutilizando singleton (uuid={cls._uuid}, calls={cls._create_count})")
        return cls._instance

class Database(metaclass=_Singleton):  # <- troque para "object" se NÃO quiser singleton
    def __init__(self):
        self.pool = get_connection_pool()  # Pool global
        self.connection = None  # Conexão atual do request

    # ---- helpers internos ----
    def _is_alive(self) -> bool:
        try:
            return self.connection is not None and self.connection.is_connected()
        except Exception:
            return False

    def connect(self):
        """Obtém uma conexão do pool."""
        if self._is_alive():
            return self.connection
        try:
            start_time = time.time()
            self.connection = self.pool.get_connection()

            if DB_DEBUG:
                elapsed = time.time() - start_time
                print(f"[Database] 🔌 Conexão obtida do pool em {elapsed:.4f}s")

            return self.connection
        except Error as e:
            raise Exception(f"Erro ao obter conexão do pool: {e}")

    def _cursor(self):
        if not self._is_alive():
            self.connect()
        return self.connection.cursor()

    def execute_query(self, query, params=None, retries=3):
        """Executa query com retry automático em caso de perda de conexão."""
        last_error = None

        for attempt in range(retries):
            try:
                # Força reconexão se não estiver viva
                if not self._is_alive():
                    if DB_DEBUG and attempt > 0:
                        print(f"[Database] 🔄 Tentando reconectar (tentativa {attempt + 1}/{retries})")
                    self.connection = None
                    self.connect()

                cur = self._cursor()
                try:
                    cur.execute(query, params or ())
                    self.connection.commit()
                    return cur
                except Error as e:
                    if self.connection and self.connection.is_connected():
                        self.connection.rollback()
                    raise
                finally:
                    cur.close()

            except Error as e:
                last_error = e
                error_code = e.errno if hasattr(e, 'errno') else None
                connection_errors = [2006, 2013, 2055]

                if error_code in connection_errors and attempt < retries - 1:
                    if DB_DEBUG:
                        print(f"[Database] ⚠️ Conexão perdida (erro {error_code}), tentando novamente...")
                    self.connection = None
                    time.sleep(0.5 * (attempt + 1))
                    continue
                else:
                    raise Exception(f"Erro ao executar query: {e}")

        raise Exception(f"Erro ao executar query após {retries} tentativas: {last_error}")

    def execute_many(self, queries):
        cur = self._cursor()
        try:
            for q in queries:
                if q and q.strip():
                    cur.execute(q)
            self.connection.commit()
        except Error as e:
            self.connection.rollback()
            raise Exception(f"Erro ao executar batch: {e}")
        finally:
            cur.close()

    def fetch_data(self, query, params=None, retries=3):
        """Busca dados com retry automático em caso de perda de conexão."""
        last_error = None
        query_start = time.time()

        for attempt in range(retries):
            try:
                # Força reconexão se não estiver viva
                if not self._is_alive():

                        print(f"[Database] 🔄 Tentando reconectar (tentativa {attempt + 1}/{retries})")
                    self.connection = None  # Force reset
                    self.connect()

                cur = self._cursor()
                try:
                    exec_start = time.time()
                    cur.execute(query, params or ())
                    exec_time = time.time() - exec_start

                    fetch_start = time.time()
                    result = cur.fetchall()
                    fetch_time = time.time() - fetch_start

                    columns = [c[0] for c in cur.description]
                    data = [dict(zip(columns, row)) for row in result]

                    total_time = time.time() - query_start

                    if DB_DEBUG:
                        print(f"[Database] 📊 Query: {exec_time:.4f}s | Fetch: {fetch_time:.4f}s | Total: {total_time:.4f}s | Rows: {len(data)}")
                        # Mostra query resumida (primeiros 100 chars)
                        query_preview = query.strip().replace('\n', ' ')[:100]
                        print(f"[Database] 📝 {query_preview}...")

                    return data
                finally:
                    cur.close()

            except Error as e:
                last_error = e
                error_code = e.errno if hasattr(e, 'errno') else None

                # Erros de conexão que devem fazer retry
                connection_errors = [2006, 2013, 2055]  # Server gone away, Lost connection, Lost connection to server at reading

                if error_code in connection_errors and attempt < retries - 1:
                    if DB_DEBUG:
                        print(f"[Database] ⚠️ Conexão perdida (erro {error_code}), tentando novamente...")
                    self.connection = None  # Force reset
                    time.sleep(0.5 * (attempt + 1))  # Backoff exponencial
                    continue
                else:
                    raise Exception(f"Erro ao buscar dados: {e}")

        raise Exception(f"Erro ao buscar dados após {retries} tentativas: {last_error}")

    def close(self):
        """Retorna a conexão ao pool (não fecha permanentemente)"""
        try:
            if self.connection and self.connection.is_connected():

        finally:
            self.connection = None

        """

            set_sql, set_params = build_set_clause(data)

            params = tuple([*set_params, id_])

    # ------------ extras úteis (opcionais) ------------

    def increment(self, where: Dict[str, Any], field: str, step: int = 1) -> int:
        """
        UPDATE ... SET `field` = `field` + step WHERE ...
        """
        where = ensure_where(where)
        if not isinstance(step, int):
            raise ValueError("step deve ser int.")
        try:
            self.db.connect()
            where_sql, where_params = build_where_clause(where)
            if not where_sql:
                raise ValueError("Condição 'where' vazia no increment.")
            sql = f"UPDATE {safe_ident(self.tabela)} SET {safe_ident(field)} = {safe_ident(field)} + %s" + where_sql
            params = tuple([step, *where_params])
            cur = self.db.execute_query(sql, params)
            return cur.rowcount if cur else 0
        except Exception as e:
            self._error('Edit', 'increment', e)
            return 0

async def list_modbus_connections(config):
    """Lista todas as conexões Modbus ativas via API."""
    async with httpx.AsyncClient(verify=False, timeout=httpx.Timeout(5.0)) as client:
        try:
            response = await client.get(f"http://{config['ip']}:{config['port']}/listConnections")
            connections_data = response.json()
            if connections_data.get('status') != 'success':
                await close_modbus_connections(config)
                raise Exception(f"[ERRO API] {connections_data.get('message')}")
            return connections_data.get('data'), time.time()
        except Exception as e:
            await close_modbus_connections(config)
            raise Exception(f"[ERRO] {e}")

                port: int

from datetime import datetime

def dt_filter(value: Any) -> str:
    """
    Filtro para formatação de datas no Jinja2.
    Converte datetime objects para string formatada no padrão brasileiro.
    """
    if value is None:
        return ""

    if isinstance(value, datetime):
        return value.strftime("%d/%m/%Y %H:%M")

    # Se não for datetime, tenta converter para string
    return str(value)


def clip_filter(value: Any, length: int = 100) -> str:
    """
    Filtro para truncar texto no Jinja2.
    Corta o texto no tamanho especificado e adiciona '...' se necessário.
    """
    if value is None:
        return ""

    text = str(value)
    if len(text) <= length:
        return text

    return text[:length] + "..."

# libs/models/utils.py
from __future__ import annotations
from typing import Any, Dict, List, Optional, Sequence, Tuple, Union
import re
import unicodedata
import json

Identifier = str
Columns = Union[str, Sequence[Identifier]]

# Identificadores simples (sem schema/pontos)
_VALID_IDENT = re.compile(r"^[A-Za-z0-9_]+$")

# caracteres “neutros” removidos SÓ para COMPARAÇÃO (nunca para salvar/mostrar na UI)
NORMALIZE_STRIP_CHARS = " -_"

# ---------------- Identifiers / Columns ----------------
def safe_ident(name: str) -> str:
    """Valida e quote um identificador simples (coluna/tabela)."""
    if not isinstance(name, str) or not _VALID_IDENT.match(name):
        raise ValueError(f"Identificador inválido: {name!r}")
    return f"`{name}`"

def columns_sql(cols: Optional[Columns]) -> str:
    """Monta a lista de colunas para SELECT."""
    if cols in (None, "*"):
        return "*"
    if isinstance(cols, (list, tuple)):
        if not cols:
            return "*"
        return ", ".join(safe_ident(c) for c in cols)
    raise TypeError("colunas deve ser '*', None ou sequência de strings.")

# ---------------- Normalização para busca/comparação ----------------
def _strip_chars_sql(expr: str, chars: str = NORMALIZE_STRIP_CHARS) -> str:
    """Gera REPLACE encadeado para remover chars no SQL."""
    for ch in chars:
        expr = f"REPLACE({expr}, '{ch}', '')"
    return expr

def norm_sql_expr(col_quoted: str, lower: bool = True, collate: str = "utf8mb4_0900_ai_ci") -> str:
    """
    Expressão SQL normalizada (comparação):
      - remove espaços/hífens/sublinhados
      - LOWER
      - collation acento/maiúscula-insensível
    """
    expr = _strip_chars_sql(col_quoted)
    if lower:
        expr = f"LOWER({expr})"
    if collate:
        expr = f"{expr} COLLATE {collate}"
    return expr

def normalize_for_search(value: Any, strip_chars: str = NORMALIZE_STRIP_CHARS, to_lower: bool = True) -> str:
    """Normaliza em Python: remove acentos e chars neutros; lower opcional."""
    s = "" if value is None else str(value)
    s = ''.join(c for c in unicodedata.normalize('NFD', s) if unicodedata.category(c) != 'Mn')
    for ch in strip_chars:
        s = s.replace(ch, '')
    return s.lower() if to_lower else s

def like_contains(v: str) -> str:    return f"%{v}%"
def like_startswith(v: str) -> str:  return f"{v}%"
def like_endswith(v: str) -> str:    return f"%{v}"

# ---------------- ORDER / LIMIT ----------------
def order_sql(order_by: Optional[str], desc: bool) -> str:
    if not order_by:
        return ""
    return f" ORDER BY {safe_ident(order_by)} {'DESC' if desc else 'ASC'} "

def limit_sql(limit: Optional[int], offset: Optional[int]) -> str:
    parts: List[str] = []
    if isinstance(limit, int) and limit > 0:
        parts.append(f" LIMIT {limit} ")
        if isinstance(offset, int) and offset >= 0:
            parts.append(f" OFFSET {offset} ")
    return "".join(parts)

# ---------------- WHERE builder ----------------
def build_where_clause(where: Dict[str, Any]) -> Tuple[str, List[Any]]:
    """
    where: {col: valor} -> '='
           {col: ('in', [1,2,3])}
           {col: ('like', '%texto%')}
           {col: ('like_norm', '%texto%')}        # normaliza col e valor
           {col: ('eq_norm', 'Texto')}
           {col: ('startswith_norm', 'Texto')}
           {col: ('endswith_norm', 'Texto')}
           {col: ('ne', valor)}                   # !=
           {col: ('gt'|'gte'|'lt'|'lte', valor)}
           {col: ('regexp', 'padrão')}

           OU sintaxe com sufixo (estilo Django ORM):
           {col__in: [1,2,3]}
           {col__gt: 10}
           {col__gte: 10}
           {col__lt: 100}
           {col__lte: 100}
           {col__ne: 'valor'}
           {col__like: '%texto%'}
           {col__like_norm: 'texto'}
           {col__eq_norm: 'texto'}
           {col__startswith_norm: 'texto'}
           {col__endswith_norm: 'texto'}
           {col__regexp: 'padrão'}
    Retorna: (" WHERE ...", [params])
    """
    if not where:
        return "", []  # <-- IMPORTANTÍSSIMO: sempre retorne tuple

    parts: List[str] = []
    params: List[Any] = []

    for key, val in where.items():
        # Determinar coluna e operador
        # Suporte para sintaxe com sufixo: col__in, col__gt, etc.
        if '__' in key:
            col_name, op_suffix = key.rsplit('__', 1)
            col = safe_ident(col_name)
            op = op_suffix.lower()
            arg = val
        elif isinstance(val, tuple) and val:
            # Sintaxe antiga: {col: ('in', [1,2,3])}
            col = safe_ident(key)
            op = str(val[0]).lower()
            arg = val[1] if len(val) > 1 else None
        else:
            # Sintaxe simples: {col: valor}
            col = safe_ident(key)
            op = "eq"  # operação padrão
            arg = val

        # Processar operador
        if op == "in":
            seq = list(arg) if isinstance(arg, (list, tuple, set)) else [arg]
            if not seq:
                parts.append("1=0")
            else:
                placeholders = ", ".join(["%s"] * len(seq))
                parts.append(f"{col} IN ({placeholders})")
                params.extend(seq)

        elif op == "like":
            parts.append(f"{col} LIKE %s")
            params.append(arg)

        elif op == "like_norm":
            expr = norm_sql_expr(col)
            parts.append(f"{expr} LIKE %s")
            params.append(normalize_for_search(arg))

        elif op == "eq" or op == "eq_norm":
            if op == "eq_norm":
                expr = norm_sql_expr(col)
                parts.append(f"{expr} = %s")
                params.append(normalize_for_search(arg))
            else:
                parts.append(f"{col} = %s")
                params.append(arg)

        elif op == "startswith_norm":
            expr = norm_sql_expr(col)
            parts.append(f"{expr} LIKE %s")
            params.append(like_startswith(normalize_for_search(arg)))

        elif op == "endswith_norm":
            expr = norm_sql_expr(col)
            parts.append(f"{expr} LIKE %s")
            params.append(like_endswith(normalize_for_search(arg)))

        elif op == "ne":
            parts.append(f"{col} <> %s")
            params.append(arg)

        elif op in ("gt", "gte", "lt", "lte"):
            cmp_map = {"gt": ">", "gte": ">=", "lt": "<", "lte": "<="}
            parts.append(f"{col} {cmp_map[op]} %s")
            params.append(arg)

        elif op == "regexp":
            parts.append(f"{col} REGEXP %s")
            params.append(arg)

        else:
            raise ValueError(f"Operador desconhecido em where: {op}")

    clause = " WHERE " + " AND ".join(parts)
    return clause, params  # <-- IMPORTANTÍSSIMO
# ---------------- FIM utils ----------------

def build_set_clause(data: Dict[str, Any]) -> tuple[str, list[Any]]:
    """
    Gera ' SET `col1`=%s, `col2`=%s ...' e os params na mesma ordem.
    """
    if not data:
        raise ValueError("SET vazio não é permitido.")
    parts: List[str] = []
    params: List[Any] = []
    for k, v in data.items():
        parts.append(f"{safe_ident(k)} = %s")
        params.append(v)
    return " SET " + ", ".join(parts), params

def build_update_sql(table: str, data: Dict[str, Any], where: Dict[str, Any]) -> tuple[str, list[Any]]:
    """
    UPDATE `table` SET ... WHERE ...
    Retorna (sql, params) com parâmetros do SET seguidos dos do WHERE.
    """
    set_sql, set_params = build_set_clause(data)
    where_sql, where_params = build_where_clause(where)
    if not where_sql:
        # proteção contra UPDATE full table acidental
        raise ValueError("Condição 'where' vazia em UPDATE.")
    sql = f"UPDATE {safe_ident(table)}" + set_sql + where_sql
    return sql, [*set_params, *where_params]

def ensure_where(where: Optional[Dict[str, Any]]) -> Dict[str, Any]:
    if not where:
        raise ValueError("Condição 'where' não pode ser vazia.")
    return where

def ensure_data(data: Optional[Dict[str, Any]]) -> Dict[str, Any]:
    if not data:
        raise ValueError("Dados de atualização (data) não podem ser vazios.")
    return data

def json_param(value: Any) -> Tuple[str, str]:
    """
    Retorna a expressão SQL e o parâmetro serializado para JSON.
    Usa %s (paramstyle do mysql-connector).
    """
    # Opção A (CAST): funciona bem no MySQL 8/5.7 recentes
    return "CAST(%s AS JSON)", json.dumps(value, ensure_ascii=False)

    # OU, se preferir máxima compatibilidade:
    # return "JSON_EXTRACT(%s, '$')", json.dumps(value, ensure_ascii=False)

def desempenho(func):
    import time
    def wrapper(*args, **kwargs):
        start_time = time.time()
        result = func(*args, **kwargs)
        end_time = time.time()
        print(f"{func.__name__} - Tempo de execução: {end_time - start_time} segundos")
        print('-' * 50)
        return result
    return wrapper

function searchVariaveis(dispositivoId, searchTerm) {
    // Debug: descomente a linha abaixo se precisar debugar
    // console.log('🔍 Busca:', dispositivoId, searchTerm);

    const resultsContainer = document.getElementById(`${dispositivoId}_search_results`);

    if (!resultsContainer) {
        console.error('❌ Container de resultados não encontrado:', `${dispositivoId}_search_results`);
        return;
    }

    const searchInput = searchTerm.trim();

    // Limpar resultados anteriores
    resultsContainer.innerHTML = '';

    if (searchInput.length < 1) {
        resultsContainer.classList.add('hidden');
        return;
    }

    // Função para normalizar texto (remover acentos e converter para minúsculas)
    const normalizeText = (text) => {
        return text
            .toLowerCase()
            .normalize('NFD')
            .replace(/[\u0300-\u036f]/g, '');
    };

    const searchNormalized = normalizeText(searchInput);
    const results = [];
    const secoesContainer = document.getElementById(`${dispositivoId}_secoes_entradas`);

    if (!secoesContainer) {
        console.error('❌ Container de seções não encontrado:', `${dispositivoId}_secoes_entradas`);
        return;
    }

    // Mapeamento de nomes de seções para português
    const secaoNomesMap = {
        'leituras': 'Leitura',
        'temperaturas': 'Temperatura',
        'alarmes': 'Alarme',
        'comandos': 'Comando',
        'vibrações': 'Vibração',
        'pressões': 'Pressão',
        'vazões': 'Vazão',
        'alarmes_reset_automatico': 'Alarme Reset Automático'
    };

    // Buscar em todas as seções
    const allSections = secoesContainer.querySelectorAll('.bg-background.rounded.p-3');

    allSections.forEach(sectionWrapper => {
        // Encontrar o container real de entradas dentro desta seção
        const containers = sectionWrapper.querySelectorAll('[id*="_leituras"], [id*="_temperaturas"], [id*="_alarmes"], [id*="_comandos"], [id*="_vibrações"], [id*="_pressões"], [id*="_vazões"]');

        containers.forEach(container => {
            const containerId = container.id;

            // Verificar se este container pertence ao dispositivo atual
            if (!containerId.startsWith(dispositivoId)) {
                return;
            }

            // Extrair tipo de entrada
            const tipoMatch = containerId.match(/_([^_]+)$/);
            if (!tipoMatch) return;

            const tipoEntrada = tipoMatch[1];

            // Buscar todas as entradas nesta seção
            const entradas = container.querySelectorAll('[id^="' + dispositivoId + '_' + tipoEntrada + '_"]');

            entradas.forEach(entradaEl => {
                // Pular o container principal
                if (entradaEl.id === containerId) return;

                const entradaId = entradaEl.id;
                const nomeInput = entradaEl.querySelector('input[data-field="nome"]');
                const tipoInput = entradaEl.querySelector('select[data-field="tipo"]');
                const enderecoInput = entradaEl.querySelector('input[data-field="endereco"]');

                if (nomeInput && nomeInput.value && nomeInput.value.trim()) {
                    const nomeOriginal = nomeInput.value;
                    const nomeNormalizado = normalizeText(nomeOriginal);
                    const tipo = tipoInput?.value || '';
                    const endereco = enderecoInput?.value || '';

                    // Busca case-insensitive, sem acentos e com palavras incompletas
                    if (nomeNormalizado.includes(searchNormalized)) {
                        results.push({
                            nome: nomeOriginal,
                            tipo: tipo,
                            endereco: endereco,
                            secao: tipoEntrada,
                            secaoNome: secaoNomesMap[tipoEntrada] || tipoEntrada.charAt(0).toUpperCase() + tipoEntrada.slice(1).replace(/_/g, ' '),
                            sectionContentId: `${dispositivoId}_${tipoEntrada}_content`,
                            entradaId: entradaId
                        });
                    }
                }
            });
        });
    });

    // Exibir resultados
    if (results.length > 0) {
        resultsContainer.classList.remove('hidden');

        results.forEach(result => {
            // Determinar cor do badge baseado no tipo de seção
            const badgeColors = {
                'Leitura': 'bg-blue-500',
                'Temperatura': 'bg-orange-500',
                'Alarme': 'bg-red-500',
                'Comando': 'bg-purple-500',
                'Vibração': 'bg-yellow-500',
                'Pressão': 'bg-green-500',
                'Vazão': 'bg-cyan-500'
            };

            const badgeColor = badgeColors[result.secaoNome] || 'bg-gray-500';

            // Escapar caracteres especiais no nome para evitar problemas no HTML
            const nomeEscapado = result.nome.replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;').replace(/"/g, '&quot;').replace(/'/g, '&#039;');        

            const resultHTML = `
                <div class="bg-surface p-2.5 rounded hover:bg-gray-700 cursor-pointer transition-colors border border-gray-700 hover:border-gray-600"
                     onclick="highlightVariavel('${result.sectionContentId}', '${result.entradaId}')">
                    <div class="flex items-start justify-between gap-2">
                        <div class="flex-1 min-w-0">
                            <div class="text-sm font-medium text-text-primary mb-1 truncate" title="${nomeEscapado}">
                                ${nomeEscapado}
                            </div>
                            <div class="flex flex-wrap items-center gap-2 text-xs">
                                <span class="${badgeColor} text-white px-2 py-0.5 rounded font-medium">
                                    ${result.secaoNome}
                                </span>
                                <span class="bg-gray-700 text-text-primary px-2 py-0.5 rounded font-medium">
                                    ${result.tipo}
                                </span>
                                <span class="text-text-muted">
                                    End: <span class="font-mono font-medium text-text-primary">${result.endereco}</span>
                                </span>
                            </div>
                        </div>
                        <span class="material-icons text-sm text-text-muted flex-shrink-0 mt-1">arrow_forward</span>
                    </div>
                </div>
            `;
            resultsContainer.insertAdjacentHTML('beforeend', resultHTML);
        });

        // Adicionar contador de resultados
        const countHTML = `
            <div class="text-xs text-text-muted text-center py-1 border-t border-gray-700 mt-1">
                ${results.length} variável${results.length !== 1 ? 'eis' : ''} encontrada${results.length !== 1 ? 's' : ''}
            </div>
        `;
        resultsContainer.insertAdjacentHTML('beforeend', countHTML);

        // Log simplificado
        console.log(`✅ Busca "${searchTerm}": ${results.length} resultado(s)`);
    } else {
        resultsContainer.classList.remove('hidden');
        resultsContainer.innerHTML = '<div class="text-xs text-text-muted italic text-center py-3">Nenhuma variável encontrada</div>';

        console.log(`❌ Busca "${searchTerm}": nenhum resultado`);
    }
}

function highlightVariavel(sectionContentId, entradaId) {
    // Expandir a seção se estiver fechada
    const sectionContent = document.getElementById(sectionContentId);
    if (sectionContent && !sectionContent.classList.contains('active')) {
        toggleSection(sectionContentId);
    }

    // Scroll até a variável e destacá-la
    const entradaEl = document.getElementById(entradaId);
    if (entradaEl) {
        entradaEl.scrollIntoView({ behavior: 'smooth', block: 'center' });

        // Adicionar efeito de destaque temporário
        entradaEl.style.transition = 'all 0.3s ease';
        entradaEl.style.backgroundColor = 'rgba(59, 130, 246, 0.3)';
        entradaEl.style.boxShadow = '0 0 0 2px rgba(59, 130, 246, 0.5)';

        setTimeout(() => {
            entradaEl.style.backgroundColor = '';
            entradaEl.style.boxShadow = '';
        }, 2000);
    }
}

function coletarDadosETeste(entradaId, tipoSecao) {
    console.log('📋 Coletando dados para teste...');

    const entradaEl = document.getElementById(entradaId);
    if (!entradaEl) {
        console.error('❌ Entrada não encontrada:', entradaId);
        return;
    }

    // Coletar dados da variável
    const nomeInput = entradaEl.querySelector('input[data-field="nome"]');
    const tipoInput = entradaEl.querySelector('select[data-field="tipo"]');
    const enderecoInput = entradaEl.querySelector('input[data-field="endereco"]');

    const nome = nomeInput?.value.trim();
    const tipo = tipoInput?.value;
    const endereco = parseInt(enderecoInput?.value);

    if (!nome || !tipo || !endereco) {
        showToast('Preencha todos os campos da variável', 'warning');
        return;
    }

    // Extrair dispositivo e usina
    const dispositivoEl = entradaEl.closest('[id*="_disp_"]');
    let usinaEl = null;

    if (dispositivoEl) {
        const match = dispositivoEl.id.match(/^(usina_\d+)_disp_/);
        if (match) {
            usinaEl = document.getElementById(match[1]);
        }
    }

    if (!usinaEl || !dispositivoEl) {
        showToast('Não foi possível identificar usina/dispositivo', 'error');
        return;
    }

    // Coletar dados da usina
    const unidadeInput = usinaEl.querySelector(`input[data-usina-id][data-field="nome"]`);
    const ipInput = usinaEl.querySelector(`input[data-usina-id][data-field="ip"]`);
    const portInput = usinaEl.querySelector(`input[data-usina-id][data-field="port"]`);

    const unidade = unidadeInput?.value.trim();
    const ip = ipInput?.value.trim();
    const port = parseInt(portInput?.value);

    if (!unidade || !ip || !port) {
        showToast('Preencha os dados da usina', 'warning');
        return;
    }

    // Coletar dados do dispositivo
    const dispositivoIdMatch = entradaId.match(/^(usina_\d+_disp_\d+)/);
    const dispositivoId = dispositivoIdMatch ? dispositivoIdMatch[1] : null;

    const conexao_ipInput = document.querySelector(`#${dispositivoId} input[data-dispositivo-id][data-field="ip"]`);
    const conexao_portInput = document.querySelector(`#${dispositivoId} input[data-dispositivo-id][data-field="port"]`);

    const conexao_ip = conexao_ipInput?.value.trim();
    const conexao_port = parseInt(conexao_portInput?.value);

    if (!conexao_ip || !conexao_port) {
        showToast('Preencha os dados do dispositivo', 'warning');
        return;
    }

    // Montar config e data
    const config = {
        entrada_id: entradaId,
        unidade,
        ip,
        port,
        tipo: tipoSecao,
        nome,
        tipo_variavel: tipo,
        endereco
    };

    const data = {
        conexao: {
            ip: conexao_ip,
            port: conexao_port
        },
        [tipoSecao]: {
            [tipo]: {
                [nome]: endereco
            }
        }
    };

    // Chamar teste
    testarLeitura(config, data);
}

function loadConfiguration(config) {
    // Limpar o container
    document.getElementById('usinasContainer').innerHTML = '';

    // Carregar cada usina
    Object.entries(config).forEach(([usinaName, usinaData]) => {
        addUsina();
        const usinaId = `usina_${usinaCounter - 1}`;
        const usinaEl = document.getElementById(usinaId);

        // Preencher dados da usina
        usinaEl.querySelector(`[data-usina-id="${usinaId}"][data-field="nome"]`).value = usinaName;
        usinaEl.querySelector(`[data-usina-id="${usinaId}"][data-field="ip"]`).value = usinaData.ip || '';
        usinaEl.querySelector(`[data-usina-id="${usinaId}"][data-field="port"]`).value = usinaData.port || '';
        usinaEl.querySelector(`[data-usina-id="${usinaId}"][data-field="table"]`).value = usinaData.table || '';

        // Remover o dispositivo padrão
        const defaultDisp = usinaEl.querySelector('[id$="_disp_0"]');
        if (defaultDisp) defaultDisp.remove();

        // Carregar dispositivos
        if (usinaData.dispositivos) {
            Object.entries(usinaData.dispositivos).forEach(([dispNome, dispData]) => {
                addDispositivo(usinaId);
                const dispId = `${usinaId}_disp_${dispositivoCounters[usinaId] - 1}`;
                const dispEl = document.getElementById(dispId);

                // Preencher dados do dispositivo
                dispEl.querySelector(`[data-dispositivo-id="${dispId}"][data-field="nome"]`).value = dispNome;
                dispEl.querySelector(`[data-dispositivo-id="${dispId}"][data-field="ip"]`).value = dispData.conexao?.ip || '';
                dispEl.querySelector(`[data-dispositivo-id="${dispId}"][data-field="port"]`).value = dispData.conexao?.port || '';

                // Carregar características
                if (dispData.caracteristicas) {
                    const caracteristicasContainer = document.getElementById(`${dispId}_caracteristicas`);
                    const firstCarInput = caracteristicasContainer.querySelector('[data-field="caracteristica"]');
                    const entries = Object.entries(dispData.caracteristicas);

                    if (entries.length > 0) {
                        // Preencher a primeira característica
                        const [firstKey, firstValue] = entries[0];
                        if (firstCarInput) {
                            firstCarInput.value = firstValue;
                            firstCarInput.previousElementSibling.textContent = firstKey;
                        }

                        // Adicionar características adicionais
                        for (let i = 1; i < entries.length; i++) {
                            const [carNome, carValor] = entries[i];
                            addCaracteristica(dispId);
                            const inputs = caracteristicasContainer.querySelectorAll('[data-field="caracteristica_nome"]');
                            const lastNomeInput = inputs[inputs.length - 1];
                            const lastValorInput = lastNomeInput.nextElementSibling;
                            lastNomeInput.value = carNome;
                            lastValorInput.value = carValor;
                        }
                    }
                }

                // Carregar leituras, temperaturas, alarmes, comandos e outras seções
                const secoesBase = ['leituras', 'temperaturas', 'alarmes', 'comandos'];
                const secoesExtras = Object.keys(dispData).filter(key =>
                    !['conexao', 'caracteristicas', ...secoesBase].includes(key) &&
                    typeof dispData[key] === 'object' &&
                    dispData[key] !== null
                );

                // Adicionar seções extras dinamicamente
                secoesExtras.forEach(secaoNome => {
                    const iconMap = {
                        'vibrações': 'vibration',
                        'pressões': 'compress',
                        'vazões': 'water_drop',
                        'alarmes_reset_automatico': 'auto_fix_high'
                    };
                    const icon = iconMap[secaoNome] || 'sensors';
                    const titulo = secaoNome.charAt(0).toUpperCase() + secaoNome.slice(1).replace(/_/g, ' ');

                    // Adicionar seção se não existir
                    if (!entradaCounters[dispId][secaoNome]) {
                        entradaCounters[dispId][secaoNome] = 0;
                        const secoesContainer = document.getElementById(`${dispId}_secoes_entradas`);
                        const secaoHTML = createEntradaSection(dispId, secaoNome, titulo, icon, ['REAL', 'INT', 'BOOLEAN']);
                        secoesContainer.insertAdjacentHTML('beforeend', secaoHTML);
                    }
                });

                // Carregar todas as seções (base + extras)
                [...secoesBase, ...secoesExtras].forEach(tipoEntrada => {
                    if (dispData[tipoEntrada]) {
                        Object.entries(dispData[tipoEntrada]).forEach(([tipo, entradas]) => {
                            Object.entries(entradas).forEach(([nomeEntrada, endereco]) => {
                                addEntrada(dispId, tipoEntrada, tipo === 'BOOLEAN' ? ['BOOLEAN'] : ['REAL', 'INT']);
                                const entradaCount = entradaCounters[dispId][tipoEntrada] - 1;
                                const entradaId = `${dispId}_${tipoEntrada}_${entradaCount}`;
                                const entradaEl = document.getElementById(entradaId);

                                if (entradaEl) {
                                    entradaEl.querySelector(`[data-entrada-id="${entradaId}"][data-field="tipo"]`).value = tipo;
                                    entradaEl.querySelector(`[data-entrada-id="${entradaId}"][data-field="nome"]`).value = nomeEntrada;
                                    entradaEl.querySelector(`[data-entrada-id="${entradaId}"][data-field="endereco"]`).value = endereco;
                                }
                            });
                        });
                    }
                });
            });
        }
    });
}

function collectFormData() {
    const config = {};

    // Coletar usinas
    const usinas = document.querySelectorAll('[id^="usina_"]');
    usinas.forEach(usinaEl => {
        const usinaId = usinaEl.id;
        const nomeInput = usinaEl.querySelector(`[data-usina-id="${usinaId}"][data-field="nome"]`);
        const nome = nomeInput ? nomeInput.value.trim() : '';

        if (!nome) return;

        config[nome] = {
            ip: usinaEl.querySelector(`[data-usina-id="${usinaId}"][data-field="ip"]`)?.value || '',
            port: parseInt(usinaEl.querySelector(`[data-usina-id="${usinaId}"][data-field="port"]`)?.value) || 0,
            table: usinaEl.querySelector(`[data-usina-id="${usinaId}"][data-field="table"]`)?.value || '',
            dispositivos: {}
        };

        // Coletar dispositivos
        const dispositivos = usinaEl.querySelectorAll(`[id^="${usinaId}_disp_"]`);
        dispositivos.forEach(dispEl => {
            const dispId = dispEl.id;
            const dispNome = dispEl.querySelector(`[data-dispositivo-id="${dispId}"][data-field="nome"]`)?.value.trim() || '';

            if (!dispNome) return;

            const dispositivoData = {
                conexao: {
                    ip: dispEl.querySelector(`[data-dispositivo-id="${dispId}"][data-field="ip"]`)?.value || '',
                    port: parseInt(dispEl.querySelector(`[data-dispositivo-id="${dispId}"][data-field="port"]`)?.value) || 0
                },
                caracteristicas: {},
                leituras: { REAL: {}, INT: {} },
                temperaturas: { REAL: {}, INT: {} },
                alarmes: { BOOLEAN: {} },
                comandos: { BOOLEAN: {} }
            };

            config[nome].dispositivos[dispNome] = dispositivoData;

            // Coletar características
            const caracteristicas = dispEl.querySelectorAll(`[data-dispositivo-id="${dispId}"][data-field="caracteristica"]`);
            caracteristicas.forEach(car => {
                const label = car.previousElementSibling?.textContent || car.placeholder;
                const value = parseFloat(car.value) || parseInt(car.value) || 0;
                if (car.value) {
                    config[nome].dispositivos[dispNome].caracteristicas[label] = value;
                }
            });

            // Características personalizadas
            const carNomes = dispEl.querySelectorAll(`[data-dispositivo-id="${dispId}"][data-field="caracteristica_nome"]`);
            carNomes.forEach(nomeEl => {
                const valorEl = nomeEl.nextElementSibling;
                const carNome = nomeEl.value.trim();
                const carValor = parseFloat(valorEl.value) || parseInt(valorEl.value) || 0;
                if (carNome && valorEl.value) {
                    config[nome].dispositivos[dispNome].caracteristicas[carNome] = carValor;
                }
            });

            // Descobrir todas as seções de entradas (base + extras)
            const secoesContainer = dispEl.querySelector(`[id="${dispId}_secoes_entradas"]`);
            const todasSecoes = [];
            if (secoesContainer) {
                // Encontrar todas as divs de seção
                secoesContainer.querySelectorAll('[id$="_leituras"], [id$="_temperaturas"], [id$="_alarmes"], [id$="_comandos"]').forEach(secaoEl => {
                    const secaoId = secaoEl.id;
                    const tipoEntrada = secaoId.replace(`${dispId}_`, '');
                    if (!todasSecoes.includes(tipoEntrada)) {
                        todasSecoes.push(tipoEntrada);
                    }
                });

                // Também buscar seções extras
                secoesContainer.querySelectorAll('[id*="_"]').forEach(el => {
                    const match = el.id.match(new RegExp(`${dispId.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')}_([^_]+)$`));
                    if (match && match[1]) {
                        const tipoEntrada = match[1];
                        if (!todasSecoes.includes(tipoEntrada) &&
                            !['dispositivos', 'caracteristicas', 'secoes', 'entradas'].includes(tipoEntrada)) {
                            todasSecoes.push(tipoEntrada);
                        }
                    }
                });
            }

            // Coletar entradas de todas as seções encontradas
            todasSecoes.forEach(tipoEntrada => {
                const entradas = dispEl.querySelectorAll(`[id^="${dispId}_${tipoEntrada}_"]`);
                if (entradas.length > 0) {
                    // Inicializar a seção se não existir
                    if (!dispositivoData[tipoEntrada]) {
                        dispositivoData[tipoEntrada] = { REAL: {}, INT: {}, BOOLEAN: {} };
                    }

                    entradas.forEach(entradaEl => {
                        const entradaId = entradaEl.id;
                        // Pular se for o container da seção
                        if (entradaId === `${dispId}_${tipoEntrada}`) return;

                        const tipo = entradaEl.querySelector(`[data-entrada-id="${entradaId}"][data-field="tipo"]`)?.value;
                        const nomeEntrada = entradaEl.querySelector(`[data-entrada-id="${entradaId}"][data-field="nome"]`)?.value.trim();
                        const endereco = parseInt(entradaEl.querySelector(`[data-entrada-id="${entradaId}"][data-field="endereco"]`)?.value);

                        if (tipo && nomeEntrada && endereco) {
                            if (!dispositivoData[tipoEntrada][tipo]) {
                                dispositivoData[tipoEntrada][tipo] = {};
                            }
                            dispositivoData[tipoEntrada][tipo][nomeEntrada] = endereco;
                        }
                    });

                    // Remover tipos vazios
                    Object.keys(dispositivoData[tipoEntrada]).forEach(tipo => {
                        if (Object.keys(dispositivoData[tipoEntrada][tipo]).length === 0) {
                            delete dispositivoData[tipoEntrada][tipo];
                        }
                    });

                    // Remover seção se estiver vazia
                    if (Object.keys(dispositivoData[tipoEntrada]).length === 0) {
                        delete dispositivoData[tipoEntrada];
                    }
                }
            });
        });
    });

    return config;
}

function showToast(message, type = 'info') {
    const toast = document.getElementById('toast');

    const toastMessage = document.getElementById('toastMessage');

    // Definir ícone e cor baseado no tipo
    const types = {
        success: { icon: 'check_circle', class: 'bg-success text-white' },
        error: { icon: 'error', class: 'bg-danger text-white' },
        warning: { icon: 'warning', class: 'bg-warning text-white' },
        info: { icon: 'info', class: 'bg-primary text-white' }
    };

    const config = types[type] || types.info;
    toastIcon.textContent = config.icon;
    toastMessage.textContent = message;

    // Remover classes anteriores e adicionar novas
    toast.className = `fixed top-4 right-4 z-50 px-6 py-3 rounded-lg shadow-lg transition-all duration-300 ${config.class}`;

    // Mostrar toast
    toast.classList.remove('hidden', 'translate-x-full');

    // Esconder após 3 segundos
    setTimeout(() => {
        toast.classList.add('translate-x-full');
        setTimeout(() => {
            toast.classList.add('hidden');
        }, 300);
    }, 3000);
}

        // Remove active de todos
        tabButtons.forEach(btn => {
            btn.classList.remove('active', 'border-primary', 'text-primary');
            btn.classList.add('border-transparent', 'text-text-muted');
        });

        // Encontra e ativa a tab específica
        const targetTab = document.getElementById(`tab-${tabId}`);
        if (targetTab) {
            targetTab.classList.add('active', 'border-primary', 'text-primary');
            targetTab.classList.remove('border-transparent', 'text-text-muted');
        }

        // Mostra/esconde forms
        if (tabId === 'registrar') {
            formOcorrencia.classList.remove('hidden');
            formResolver.classList.add('hidden');
        } else if (tabId === 'resolver') {
            formOcorrencia.classList.add('hidden');
            formResolver.classList.remove('hidden');
            return carregarOcorrenciasAbertas(); // Retorna a promise
        }

    function formatarDataTimeLocal(date) {
        const year = date.getFullYear();
        const month = String(date.getMonth() + 1).padStart(2, '0');
        const day = String(date.getDate()).padStart(2, '0');
        const hours = String(date.getHours()).padStart(2, '0');
        const minutes = String(date.getMinutes()).padStart(2, '0');
        return `${year}-${month}-${day}T${hours}:${minutes}`;
    }

    async function carregarOcorrenciasAbertas() {
        try {
            // Busca apenas ocorrências que requerem ação e ainda não foram resolvidas
            const response = await fetch('/api/ocorrencias?status=aberta,em_andamento&requer_acao=true');
            const result = await response.json();

            const select = document.getElementById('ocorrencia_resolver');
            select.innerHTML = '<option value="">Selecione uma ocorrência que requer ação</option>';

            if (result.success && result.data) {
                // Filtra apenas ocorrências com requer_acao=true
                const ocorrenciasComAcao = result.data.filter(oc => oc.requer_acao === true || oc.requer_acao === 1);

                if (ocorrenciasComAcao.length === 0) {
                    const option = document.createElement('option');
                    option.value = '';
                    option.textContent = 'Nenhuma ocorrência requer ação no momento';
                    option.disabled = true;
                    select.appendChild(option);
                } else {
                    ocorrenciasComAcao.forEach(oc => {
                        const option = document.createElement('option');
                        option.value = oc.id;
                        option.textContent = `#${oc.id} - ${oc.tipo} - ${oc.unidade} (${oc.status})`;
                        option.dataset.ocorrencia = JSON.stringify(oc);
                        select.appendChild(option);
                    });
                }
            }
        } catch (error) {
            console.error('Erro ao carregar ocorrências:', error);
        }
    }

        <script>
            // Dados originais das ocorrências
            const ocorrenciasOriginais = document.querySelectorAll('.ocorrencia-item');

            function aplicarFiltros() {
                const filtroTipo = document.getElementById('filtro-tipo').value.toLowerCase();
                const filtroCategoria = document.getElementById('filtro-categoria').value.toLowerCase();
                const filtroStatus = document.getElementById('filtro-status').value.toLowerCase();
                const filtroUnidade = document.getElementById('filtro-unidade').value.toLowerCase();

                let contador = 0;

                ocorrenciasOriginais.forEach(item => {
                    const tipo = item.dataset.tipo.toLowerCase();
                    const categoria = item.dataset.categoria.toLowerCase();
                    const status = item.dataset.status.toLowerCase();
                    const unidade = item.dataset.unidade.toLowerCase();
                    const tags = item.dataset.tags.toLowerCase();

                    // Verificar se passa por todos os filtros
                    const passaTipo = !filtroTipo || tipo.includes(filtroTipo);
                    const passaCategoria = !filtroCategoria || categoria.includes(filtroCategoria);
                    const passaStatus = !filtroStatus || status.includes(filtroStatus);
                    const passaUnidade = !filtroUnidade || unidade.includes(filtroUnidade) || tags.includes(filtroUnidade);

                    if (passaTipo && passaCategoria && passaStatus && passaUnidade) {
                        item.style.display = 'flex';
                        contador++;
                    } else {
                        item.style.display = 'none';
                    }
                });

                // Atualizar contador
                document.getElementById('contador-filtrados').textContent = `${contador} ocorrências`;

                // Mostrar/ocultar mensagem de vazia
                const mensagemVazia = document.getElementById('mensagem-vazia');
                if (mensagemVazia) {
                    mensagemVazia.style.display = contador === 0 ? 'block' : 'none';
                }
            }

            function limparFiltros() {
                document.getElementById('filtro-tipo').value = '';
                document.getElementById('filtro-categoria').value = '';
                document.getElementById('filtro-status').value = '';
                document.getElementById('filtro-unidade').value = '';
                aplicarFiltros();
            }

            // Assumir responsabilidade de uma ocorrência
            async function assumirResponsabilidade(ocorrenciaId) {
                if (!confirm('Deseja assumir a responsabilidade por esta ocorrência?')) {
                    return;
                }

                try {
                    const response = await fetch(`/api/ocorrencias/${ocorrenciaId}/assumir`, {
                        method: 'POST',
                        headers: {
                            'Content-Type': 'application/json',
                        }
                    });

                    const result = await response.json();

                    if (result.success) {
                        alert('Responsabilidade assumida com sucesso!');
                        window.location.reload();
                    } else {
                        alert('Erro ao assumir responsabilidade: ' + (result.error || 'Erro desconhecido'));
                    }
                } catch (error) {
                    console.error('Erro:', error);
                    alert('Erro ao assumir responsabilidade. Tente novamente.');
                }
            }

            // Aplicar filtros ao pressionar Enter no campo de unidade
            document.getElementById('filtro-unidade').addEventListener('keypress', function(e) {
                if (e.key === 'Enter') {
                    aplicarFiltros();

import asyncio

@socketio.on('testar_leitura')
def handle_testar_leitura(payload):
    """Handler para testar leitura de variável do CLP"""
    config = payload.get('config', {})
    data = payload.get('data', {})
    entrada_id = payload.get('entrada_id')

    print(f"🧪 Recebido pedido de teste:")
    print(f"   Unidade: {config.get('unidade')}")
    print(f"   Tipo: {config.get('tipo')}")

    try:
        # Executar função async em thread separada
        loop = asyncio.new_event_loop()
        asyncio.set_event_loop(loop)
        resultado, tempo = loop.run_until_complete(get_data(config, data))
        loop.close()

        print(f"📊 Resultado da API: {resultado}")

        # Extrair o valor da resposta
        tipo_secao = config.get('tipo')  # leituras, temperaturas, etc

        if resultado:
            # Navegar pela estrutura de resposta para pegar o valor
            for tipo_dado, valores in resultado.items():  # REAL, INT, BOOLEAN
                for nome_var, valor in valores.items():
                    # Emitir sucesso
                    emit('resultado_teste_leitura', {
                        'entrada_id': entrada_id,
                        'status': 'success',
                        'nome': nome_var,
                        'tipo': tipo_dado,
                        'valor': valor,
                        'tempo': f"{tempo:.3f}s",
                        'message': 'Leitura realizada com sucesso'
                    })
                    return

        # Se não encontrou valor
        emit('resultado_teste_leitura', {
            'entrada_id': entrada_id,
            'status': 'error',
            'nome': 'desconhecida',
            'message': 'Nenhum valor retornado pela API'
        })

    except Exception as e:
        print(f"❌ Erro ao testar leitura: {e}")
        import traceback
        traceback.print_exc()
        emit('resultado_teste_leitura', {
            'entrada_id': entrada_id,
            'status': 'error',
            'nome': 'desconhecida',
            'message': f'Erro: {str(e)}'
        })

if DEV_RELOAD:
    from watchdog.observers import Observer
    from watchdog.events import FileSystemEventHandler

    class FileChangeHandler(FileSystemEventHandler):
        def __init__(self, socketio):
            self.socketio = socketio
            self.last_modified = {}

        def on_modified(self, event):
            if event.is_directory:
                return
            if event.src_path.endswith(('.html', '.css', '.js')):
                if event.src_path in self.last_modified and time.time() - self.last_modified[event.src_path] < 1:
                    return
                self.last_modified[event.src_path] = time.time()
                print(f"Arquivo modificado: {event.src_path}")
                self.socketio.emit("file_changed", {"path": event.src_path})

    def setup_file_watcher(socketio):
        handler = FileChangeHandler(socketio)
        observer = Observer()
        observer.schedule(handler, BASE_DIR, recursive=True)
        observer.start()
        return observer
